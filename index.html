<script>
let slide = 0;
let imgs = [];
let dots = [];
let score = 0;
const text =
"Thank you for all the laughs, fights, teasing and crazy memories. You make life more fun
just by being my sister. I hope your birthday is full of happiness, smiles and amazing
memories. ";
/* START */
function start() {
document.getElementById("start")
.classList.add("hidden");
document.getElementById("main")
.classList.remove("hidden");
document.getElementById("game")
.classList.remove("hidden");
imgs = [
...document.querySelectorAll(".gallery img")
];
dots = imgs.map((_, i) => {
let d = document.createElement("span");
d.className =
"dot" + (i === 0 ? " active" : "");
document.getElementById("dots")
.appendChild(d);
return d;
});
/* Confetti/hearts */
for (let i = 0; i < 60; i++) {
setTimeout(heart, i * 50);
}
/* Slideshow */
setInterval(next, 3500);
}
/* SHOW PHOTO */
function show() {
imgs.forEach((img, i) => {
img.classList.toggle(
"active",
i === slide
);
});
dots.forEach((dot, i) => {
dot.classList.toggle(
"active",
i === slide
);
});
}
/* NEXT PHOTO */
function next() {
if (!imgs.length) return;
slide =
(slide + 1) % imgs.length;
show();
}
/* PREVIOUS PHOTO */
function prev() {
if (!imgs.length) return;
slide =
(slide - 1 + imgs.length) % imgs.length;
show();
}
/* FALLING HEARTS */
function heart() {
let h =
document.createElement("div");
h.className = "heart";
h.textContent =
[" "," "," "," "," "]
[Math.floor(Math.random() * 5)];
h.style.left =
Math.random() * 100 + "vw";
h.style.animationDuration =
3 + Math.random() * 3 + "s";
document.body.appendChild(h);
setTimeout(() => h.remove(), 6500);
}
/* GAME */
function catchHeart() {
score++;
document.getElementById("score")
.textContent =
`Hearts caught: ${score} / 5`;
if (score >= 5) {
document.getElementById("game")
.classList.add("hidden");
document.getElementById("message")
.classList.remove("hidden");
typeText();
} else {
moveHeart();
}
}
/* MOVE HEART */
function moveHeart() {
let b =
document.getElementById("catch");
b.style.display = "block";
b.style.left =
Math.random() * 85 + "%";
b.style.top =
Math.random() * 70 + "%";
}
/* TYPING MESSAGE */
function typeText() {
let i = 0;
let el =
document.getElementById("typing");
el.textContent = "";
let timer =
setInterval(() => {
el.textContent += text[i++];
if (i >= text.length) {
clearInterval(timer);
}
}, 28);
}
/* FINAL SURPRISE */
function finish() {
document.getElementById("message")
.classList.add("hidden");
document.getElementById("final")
.classList.remove("hidden");
for (let i = 0; i < 80; i++) {
setTimeout(
heart,
i * 35
);
}
}
/* GAME HEART ON LOAD */
document.addEventListener(
"DOMContentLoaded",
moveHeart
);
</script>
